<StoryHeading of={CarouselStories.SlideNoControls}></StoryHeading>

<CanvasTabs htmlOf={CarouselStories.SlideNoControlsHtml} of={CarouselStories.SlideNoControls} />

<CanvasTabs htmlOf={CarouselStories.ProductHtml} of={CarouselStories.Product} />
