Options can be declared as HTML markup with slotted `ds-select-option` (and `ds-select-optgroup`) elements —
useful for backend-only apps and easier migration — or set programmatically via the `options` / `optionGroups` prop.

<HTMLFramework>

```html
<ds-select id="select-country" label="Country" description="Select your country of residence"></ds-select>

<script>
  document.querySelector('#select-country').options = [
    { label: 'Switzerland', value: 'ch' },
    { label: 'Germany', value: 'de' },
    { label: 'Italy', value: 'it' },
  ]
</script>
```

</HTMLFramework>

<ReactFramework>

```tsx
import { DsSelect } from '@baloise/ds-react'

const countryOptions = [
  { label: 'Switzerland', value: 'ch' },
  { label: 'Germany', value: 'de' },
  { label: 'Italy', value: 'it' },
]

export function CountrySelect() {
  return <DsSelect
    label="Country"
    description="Select your country of residence"
    options={countryOptions} />
}
```

</ReactFramework>

<AngularFramework>

```html
<!-- app.component.html -->
<ds-select
  label="Country"
  description="Select your country of residence"
  [options]="countryOptions"></ds-select>
```

```ts
// app.component.ts
import { Component } from '@angular/core'

@Component({
  selector: 'app-country-select',
  templateUrl: './app.component.html',
})
export class AppComponent {
  countryOptions = [
    { label: 'Switzerland', value: 'ch' },
    { label: 'Germany', value: 'de' },
    { label: 'Italy', value: 'it' },
  ]
}
```

</AngularFramework>
